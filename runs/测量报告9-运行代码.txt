def parse(s):
    rows = []
    for line in s.strip().splitlines():
        line = line.split("#")[0].strip()
        if line:
            rows.append([float(t) for t in line.replace(",", " ").split()])
    return np.array(rows) if len(rows) == 17 else None

def load(key, name):
    A = parse(key)
    if A is not None:
        return A, "float64(粘贴)"
    import bpy
    k = name.replace(" ", "").lower()
    nm = next(n for n in bpy.data.objects.keys() if n.replace(" ", "").lower() == k)
    ob = bpy.data.objects[nm]; M = ob.matrix_world
    return np.array([(M @ v.co)[:] for v in ob.data.vertices]), "float32(Blender回退)"

IDX2, IDX11 = 2, 11                 # 0基索引（报告6生效版本）
SQ83 = math.sqrt(8.0 / 3.0)         # 归一化→模型单位
SPECTRUM = [("对(2,11)",     Fraction(40, 27)),
            ("对(3,10)",     Fraction(4, 3)),
            ("对(4,6)(7,9)", Fraction(8, 9))]

def analyze(V, tag):
    logS(f"\n========== {tag} ==========")
    c0i = int(np.argmin([np.linalg.norm(V - x, axis=1).max() for x in V]))
    c0 = V[c0i]; rel = V - c0
    Rc = float(np.linalg.norm(rel, axis=1).max())
    w = rel.mean(axis=0); nrm = float(np.linalg.norm(w))
    nn = V[IDX11] - V[IDX2]; nn = nn / np.linalg.norm(nn)
    if nn[0] < 0: nn = -nn          # A.7 定向：n_x > 0
    logS(f"中心顶点 idx={c0i}; R_c = {Rc:.15f}; |w|/R_c = {nrm/Rc:.15f}")
    logS(f"n = ({nn[0]:+.15f}, {nn[1]:+.15f}, {nn[2]:+.15f})")
    logS(f"  闭式核对: n_x-2√2/3 = {nn[0]-2*math.sqrt(2)/3:.3e}; "
         f"n_y+1/3 = {nn[1]+1/3:.3e}; n_z = {nn[2]:.3e}")
    nw = float(nn @ w); dp = rel @ nn
    # ---- δ 双通道（R1：直取通道剔零档）----
    delta_hat_a = abs(nw) * 17.0 / 5.0
    nz = np.sort(np.abs(dp[np.abs(dp) > 1e-3]))[:5]
    delta_hat_b = float(nz.mean())
    delta_model = delta_hat_a * SQ83
    logS(f"n·w = {nw:+.15f}")
    logS(f"δ̂(恒等式)  = {delta_hat_a:.15f}")
    logS(f"δ̂(直接5类) = {delta_hat_b:.15f}   双通道差 = {abs(delta_hat_a-delta_hat_b):.2e}  ← 应<1e-7")
    logS(f"δ(模型单位) = {delta_model:.15f}")
    # ---- 镜距谱解码（R2：降序分段）----
    logS("[镜距谱解码·模型单位]")
    vals = sorted(np.abs(dp * SQ83), reverse=True)
    SEG = {SPECTRUM[0][0]: (0, 2), SPECTRUM[1][0]: (2, 4), SPECTRUM[2][0]: (4, 8)}
    for name, frac in SPECTRUM:
        a, b = SEG[name]
        m = sum(vals[a:b]) / (b - a)
        e = float(frac)
        logS(f"  {name}: 实测 {m:.12f} vs 闭式 {e:.12f} (= {frac})  相对残差 = {abs(m-e)/e:.1e}")
    logS(f"  五重类: 实测 {delta_model:.12f} vs [目标δ] ← 本报告识别对象")
    return dict(rc=Rc, w=nrm/Rc, nw=nw, d_hat=delta_hat_a,
                d_model=delta_model, ch=abs(delta_hat_a - delta_hat_b))

A = load(COORDS_PLUS_Z,  "+z闭合构型"); Va, src_a = A

# ---- A.8：阈值绑定输入精度 ----

DMAX    = 10**6 if FLOAT64 else 5000

w_ratio = float(0.5 * (r1["w"] + r2["w"]))

FAMILY = (17, 19, 27, 3, 9, 81, 729, 531441)
def family_ok(den):
    d = abs(int(den))
    if d == 1: return True
    for f in FAMILY:
        while d % f == 0: d //= f
    return d == 1

results = {}
BASIS = [sqrt(k) for k in (2,3,5,7,13,17,19,38,57,86,89,257)] + [(1+sqrt(5))/2] if SYMPY_OK else None

def recognize(_v, _t):
    x = float(_v); tag = str(_t)
    logS(f"\n----- 目标: {tag} = {x:.15f} -----")
    # 通道1：有理（A.8 反假象：命中须过结构族检验）
    fr = Fraction(x).limit_denominator(DMAX)
    err = abs(float(fr) - x) / max(abs(x), 1e-12)
    ok = err < REL_TOL and family_ok(fr.denominator)
    logS(f"[有理·A.8] {fr}  相对残差 = {err:.1e}  结构族 = {family_ok(fr.denominator)}"
         + ("   ← 定案" if ok else ("   ← 命中但分母出族，判废" if err < REL_TOL else "")))
    if ok: results[tag] = str(fr)
    # 通道2：√m/k 定向扫描
    hits = []; v2 = x * x
    for k in range(1, 5001):
        m = v2 * k * k; mi = int(round(m))
        if mi > 1 and abs(m - mi) < 1e-13 * m and abs(math.sqrt(mi)/k - x)/max(abs(x),1e-12) < REL_TOL:
            hits.append((mi, k))
    logS("[√m/k 扫描] " + ("命中: " + ", ".join(f"√{m}/{k}" for m, k in hits[:6]) if hits else "无命中"))
    for m, k in hits[:3]: results.setdefault(tag, f"√{m}/{k}")
    # 通道3：nsimplify（R3：参数 full，非 full_precision）
    if SYMPY_OK:
        for mode, y in [("直接", x), ("平方", x*x), ("×17", x*17), ("×27", x*27), ("×17·27", x*17*27)]:
            try:
                # ① 禁止纯有理数答案，强制代数识别（含 BASIS 中的根式/黄金比才可能命中）
                c = nsimp(y, BASIS, tolerance=1e-20, rational=False, full=False)
                cv = float(c.evalf(30)); r = abs(cv - y) / max(abs(y), 1e-12)
                # ② 命中判定加结构检验：候选须含根式成分
                hit = (r < REL_TOL) and (c.free_symbols or c.has(sqrt(2)) or any(c.has(s) for s in BASIS))
                logS(f"[nsimplify·{mode}] {tag}: {c}  相对残差 = {r:.1e}" + ("   ← 定案" if hit else ""))
                if hit:
                    suffix = ("（平方还原）" if mode == "平方"
                              else (f"（/{mode[1:]}还原）" if mode.startswith("×") else ""))
                    results.setdefault(tag, str(c) + suffix)
            except Exception as e:
                logS(f"[nsimplify·{mode}] {tag}: 失败 — {e}")
    else:
        logS("[nsimplify] sympy 不可用，通道3跳过。")

recognize(d_model, "δ(模型单位)")
recognize(w_ratio, "|w|/R_c")
