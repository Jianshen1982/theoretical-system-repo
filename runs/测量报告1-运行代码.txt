import bpy, bmesh, math
import numpy as np
from collections import Counter

# ---- 输出重定向到文件 ----
OUT = r"D:\物理数学\数学\理想球体的有限不可实现性\测量报告.txt"
sys_out = open(OUT, "w", encoding="utf-8")

def log(*args):
    print(*args, file=sys_out)

TOL = 0.005

SIGNATURES = [
    ("锁定",  2.000000),
    ("档I",   8*math.sqrt(6)/9),        # 2.177324
    ("档II",  2.241480),
    ("档III", math.sqrt(1292/243)),     # 2.305835
    ("档?",   2.592600),
]

def get_verts(ob):
    return np.array([v.co[:] for v in ob.data.vertices], dtype=np.float64)

def bucket(L):
    for name, ref in SIGNATURES:
        if abs(L - ref) < TOL:
            return name, ref
    return None, None

def angle_defects(bm):
    out = {}
    for v in bm.verts:
        s = 0.0
        for f in v.link_faces:
            pts = [x.co.copy() for x in f.verts]
            i = f.verts.index(v)
            u = pts[(i+1) % 3] - pts[i]
            w = pts[(i+2) % 3] - pts[i]
            s += u.angle(w) * 180 / math.pi
        out[v.index] = (360.0 - s, s)
    return out

for NAME in ["Solid.000", "Solid.001", "Solid.002"]:
    ob = bpy.data.objects[NAME]
    bm = bmesh.new()
    bm.from_mesh(ob.data)
    bm.verts.ensure_lookup_table(); bm.edges.ensure_lookup_table()
    verts = get_verts(ob)
    V, E, F = len(bm.verts), len(bm.edges), len(bm.faces)
    closed = (V - E + F == 2)

    log("="*62)
    log(f"{NAME}:  V={V}  E={E}  F={F}  Euler={V-E+F}  ->  "
        f"{'闭合' if closed else '开放'}")

    tally = Counter()
    log("[棱长普查]")
    rows = []
    for e in bm.edges:
        a, b = [v.index for v in e.verts]
        L = np.linalg.norm(verts[a] - verts[b])
        name, ref = bucket(L)
        rows.append((tuple(sorted((a, b))), L, name, ref,
                     None if ref is None else (L - ref)))
    for (a, b), L, name, ref, dev in sorted(rows, key=lambda r: r[1]):
        tag = (f"{name} ref={ref:.6f} 偏差={dev:+.2e}" if name
               else "** 未入档 **")
        log(f"  棱 {a:>2}-{b:<2}  L = {L:.6f}   {tag}")
        tally[name if name else "未入档"] += 1
    log("  归档统计:", dict(tally))

    if closed:
        defs = angle_defects(bm)
        s = sum(d for d, _ in defs.values())
        log("[角亏表]")
        for i in sorted(defs):
            d, sigma = defs[i]
            log(f"  顶点{i:>2}:  Σ = {sigma:8.3f}°   "
                f"角亏 = {d:8.3f}°   δ = {d-60:+8.3f}")
        log(f"  合计 = {s:.3f}°   (应为 720.000)")
    else:
        log("[开态窗口]  开放结构，角亏跳过")
        log("  请从棱长普查确认窗口五棱是否持存 2r")

    bm.free()

# ---- 分支镜像检验 ----
v1 = get_verts(bpy.data.objects["Solid.001"])
v2 = get_verts(bpy.data.objects["Solid.002"])
if len(v1) == len(v2):
    refl = v1.copy(); refl[:, 2] *= -1
    D = np.linalg.norm(refl[:, None, :] - v2[None, :, :], axis=2)
    match = D.min(axis=1)
    log("="*62)
    log(f"[镜像检验 z→-z]  逐点最近距离: max = {match.max():.6f}, "
        f"mean = {match.mean():.6f}")
    log("  " + ("匹配成功：两分支互为镜像" if match.max() < 1e-3
              else "超出容差：请检查两分支顶点序是否同源"))

sys_out.close()
print("报告已写入:", OUT)   # 这一行会出现在Blender信息栏
