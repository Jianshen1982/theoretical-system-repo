def load_bpy(name):
    import bpy
    k = name.replace(" ","").lower()
    nm = next(n for n in bpy.data.objects.keys() if n.replace(" ","").lower()==k)
    ob = bpy.data.objects[nm]; M = ob.matrix_world
    return np.array([(M @ v.co)[:] for v in ob.data.vertices], dtype=np.float64)

def center_idx(V): return int(np.argmin([np.linalg.norm(V-x,axis=1).max() for x in V]))

def classify(V):
    iu = np.triu_indices(17,1); P,Q = iu
    d2 = ((V[P]-V[Q])**2).sum(1)
    lab = np.argmin(np.abs(d2[:,None]-TGT[None,:]),axis=1)
    rel = np.abs(d2-TGT[lab])/TGT[lab]; inl = rel < 1e-3
    cnt = [int((lab[inl]==k).sum()) for k in range(3)]
    logS(f"  内点={int(inl.sum())} 计数={cnt} 残差={rel[inl].max():.1e}")
    return (P[inl],Q[inl],TGT[lab[inl]]) if (int(inl.sum())==51 and cnt==[45,4,2]) else None

def lm_solve(V0,P,Q,t,c0i,iters=300):
    X = V0.copy()
    def resid(X):
        r = ((X[P]-X[Q])**2).sum(1)-t
        g = 1e2*np.concatenate([[*X[c0i]], np.cross(X[2]-X[c0i],X[3]-X[c0i])[:2]])
        return np.concatenate([r,g])
    r = resid(X); lam = 1e-8; ne = len(t)
    for it in range(iters):
        J = np.zeros((len(r),X.size))
        for k in range(X.size):
            e = np.zeros(X.size); e[k]=1e-7
            J[:,k] = (resid(X+e.reshape(17,3))-r)/1e-7
        JTJ,JTr = J.T@J, J.T@r
        for _ in range(40):
            try: dx = np.linalg.solve(JTJ+lam*np.eye(JTJ.shape[0]),-JTr)
            except np.linalg.LinAlgError: lam*=10; continue
            Xn = X+dx.reshape(17,3); rn = resid(Xn)
            if (rn**2).sum() < (r**2).sum():
                X,r = Xn,rn; lam = max(lam*0.3,1e-14); break
            lam *= 10
        else: break
        if math.sqrt((r[:ne]**2).mean()) < 1e-15: break
    return X, math.sqrt((r[:ne]**2).mean())

edges = None
for tag,obj in [("+Z","+z闭合构型"),("-Z","-z闭合构型")]:
    logS(f"\n############ {tag} 支 ############")
    V0 = load_bpy(obj); c0i = center_idx(V0)
    if c0i != 1: logS("中心顶点≠1，中止"); continue

    sol = None; cand = None

        sol = classify(V0c*c)
        if sol: cand = c; break
    if sol is None: logS("[中止] 无标度成立"); continue
    P,Q,t = sol; edges = (P,Q,t)
    Xm,rms = lm_solve(V0c*cand,P,Q,t,c0i); Xm = Xm - Xm[c0i]
    logS(f"[LM] d² RMS = {rms:.2e}")
    # R8''' 尺度修复：对齐目标是同尺度种子
    H = Xm.T @ (V0c*cand); U,S,Vt = np.linalg.svd(H)
    det = float(np.linalg.det(U@Vt))
    if det < 0: U[:,-1] *= -1
    R = U@Vt
    res = float(np.linalg.norm(Xm@R - V0c*cand, axis=1).max())
    logS(f"[R8''' 全等性·修复后] 残差 = {res:.2e} (det={det:+.0f}) ← 应~1e-5 ⇒ 种子≅LM实现，v5的'第二实现'系尺度bug假警报")
    Xa = Xm @ R
    # R17 五重邻接普查
    d5 = np.linalg.norm(Xa - Xa[0], axis=1)
    nb = [i for i in range(17) if i != 0 and abs(d5[i]-SQ83) < 1e-6]
    logS(f"[R17 五重邻接] 与塌缩点距离=√(8/3) 的顶点 = {nb}（预期 [1,11,12] 或镜像 [1,2,12]）")
    # R20 n-标定：把镜像法向精确旋到闭式方向
    nn = Xa[11]-Xa[2]; nn /= np.linalg.norm(nn)
    if nn[0] < 0: nn = -nn
    v = np.cross(nn,NCF); c = float(nn@NCF); sn = np.linalg.norm(v)
    if sn > 1e-15:
        vx = np.array([[0,-v[2],v[1]],[v[2],0,-v[0]],[-v[1],v[0],0]])
        Xa = Xa @ (np.eye(3) + vx + vx@vx*((1-c)/sn**2))
    dp = Xa @ NCF
    logS(f"[R20] 标定后: 零档 max|dp| = {max(abs(dp[i]) for i in ZERO):.2e}; "
         f"δ = {abs(dp[FIVE]).mean():.15f}; n·w = {float(NCF@Xa.mean(0)):+.12f}")
    fp = os.path.join(OUTDIR, f"附录E_{tag}.txt")
    np.savetxt(fp, Xa, fmt="%.17e")
    logS(f"[导出·n标定框架·模型单位] {fp}")
