# -*- coding: utf-8 -*-
# 三镜像链测量 v6: +Z的x/y/z镜像 分别 vs -Z闭合构型
# 每链输出: Kabsch旋转R(角度+轴) / 残差RMSD / 复合非真操作Q=R·P的反映轴+反映角
# 理论对照: arctan(4√2/7)=38.94244°, π-2φ=102.11511°, arcsin(1/3)=19.47122°
import bpy, os, datetime
import numpy as np
from math import degrees, acos, atan, sqrt, asin

NAME_C = "-Z闭合构型"
NAMES_MIR = ["+Z闭合构型X镜像", "+Z闭合构型Y镜像", "+Z闭合构型Z镜像"]
PAXES = {"+Z闭合构型X镜像": np.diag([-1.0,1.0,1.0]),
         "+Z闭合构型Y镜像": np.diag([1.0,-1.0,1.0]),
         "+Z闭合构型Z镜像": np.diag([1.0,1.0,-1.0])}
OUT_DIR = r"D:\理论体系\测量代码 & 测量结果"

def group_points(prefix):
    objs = [o for o in bpy.data.objects if o.type=='MESH' and o.name.startswith(prefix)]
    if not objs: return None
    dg = bpy.context.evaluated_depsgraph_get()
    raw = []
    for o in objs:
        oe = o.evaluated_get(dg); me = oe.to_mesh(); mw = oe.matrix_world
        raw += [list(mw @ v.co) for v in me.vertices]
        oe.to_mesh_clear()
    pts = np.array(raw)
    order = np.argsort(np.linalg.norm(pts - pts.mean(0), axis=1))
    nodes = []
    for p in pts[order]:
        if nodes and min(np.linalg.norm(np.array(n)-p) for n in nodes) < 1e-4:
            continue
        nodes.append(list(p))
    return np.array(nodes)

def kabsch_R(A, B):
    N = len(A); assert N == len(B), f"节点数不等 {N} vs {len(B)}"
    ra = np.argsort(np.linalg.norm(A - A.mean(0), axis=1))
    rb = np.argsort(np.linalg.norm(B - B.mean(0), axis=1))
    idx = np.empty(N, dtype=int); idx[ra] = rb
    best_R, best = None, np.inf
    for _ in range(60):
        A0 = A - A.mean(0); B0 = B[idx] - B[idx].mean(0)
        U, S, Vt = np.linalg.svd(A0.T @ B0)
        d = np.sign(np.linalg.det(Vt.T @ U.T))
        R = Vt.T @ np.diag([1,1,d]) @ U.T
        Aa = A0 @ R.T
        rmsd = np.sqrt(((Aa - B0)**2).sum(1).mean())
        if rmsd < best: best, best_R = rmsd, R.copy()
        D = np.linalg.norm(Aa[:,None,:] - B[None,:,:], axis=2)
        new_idx = np.full(N, -1); used = set()
        for i, j in np.dstack(np.unravel_index(np.argsort(D, axis=None), D.shape))[0]:
            if new_idx[i] == -1 and j not in used:
                new_idx[i] = j; used.add(j)
        if np.array_equal(new_idx, idx): break
        idx = new_idx
    return best_R, best

def axis_angle(R):
    c = max(-1.0, min(1.0, (np.trace(R)-1.0)/2.0))
    th = acos(c)
    if th < 1e-8: return 0.0, np.array([0,0,1.0])
    if abs(np.pi-th) < 1e-6:
        w_, v_ = np.linalg.eigh((R+np.eye(3))/2.0)
        return th, v_[:, -1]
    ax = np.array([R[2,1]-R[1,2], R[0,2]-R[2,0], R[1,0]-R[0,1]])
    return th, ax/np.linalg.norm(ax)

phi_pred = degrees(atan(4*sqrt(2)/7))     # 38.94244
q_pred   = 180.0 - 2*phi_pred             # 102.11511
magic    = degrees(asin(1/3))             # 19.47122

XC = group_points(NAME_C)
lines = []
def w(s=""): lines.append(s); print(s)

w("="*66)
w("三镜像链测量 v6  " + datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S"))
w("="*66)
w(f"理论角: 2φ={phi_pred:.5f}°  π-2φ={q_pred:.5f}°  魔角={magic:.5f}°")

for nm in NAMES_MIR:
    XM = group_points(nm)
    w("-"*66)
    if XM is None or XC is None or len(XM) != len(XC):
        w(f"[{nm}] ✗ 取点失败或节点数不符 (M={None if XM is None else len(XM)}, "
          f"C={None if XC is None else len(XC)}),跳过")
        continue
    R, rms = kabsch_R(XM, XC)
    th, ax = axis_angle(R)
    w(f"[{nm}]")
    w(f"  真旋转: 角度={degrees(th):.7f}°  残差={rms:.2e}")
    w(f"  转轴 = ({ax[0]:+.7f}, {ax[1]:+.7f}, {ax[2]:+.7f})")
    # 复合非真操作 Q = R·P_axis: 特征值-1方向=反映轴, 反映角 cosθ=(trQ-1)/2
    Q = R @ PAXES[nm]
    wv, vv = np.linalg.eig(Q)
    k = np.argmin(np.abs(wv + 1))
    aQ = np.real(vv[:, k]); aQ /= np.linalg.norm(aQ)
    if aQ[np.argmax(np.abs(aQ))] < 0: aQ = -aQ
    cQ = max(-1.0, min(1.0, (np.trace(Q).real - 1.0)/2.0))
    w(f"  复合Q=R·P: 反映轴=({aQ[0]:+.5f},{aQ[1]:+.5f},{aQ[2]:+.5f})  "
      f"反映角={degrees(acos(cQ)):.7f}°")
    w(f"  对照: |θ-2φ|={abs(degrees(th)-phi_pred):.5f}°  "
      f"|θ_Q-(π-2φ)|={abs(degrees(acos(cQ))-q_pred):.5f}°")

os.makedirs(OUT_DIR, exist_ok=True)
stamp = datetime.datetime.now().strftime("%Y-%m-%d-%H-%M-%S")
with open(os.path.join(OUT_DIR, f"三镜像链v6+{stamp}.txt"), "w", encoding="utf-8") as f:
    f.write("\n".join(lines))
print("已保存")
